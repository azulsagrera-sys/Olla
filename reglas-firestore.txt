rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /hogares/{codigo} {
      allow get: if request.auth != null && codigo.size() >= 24;
      allow create, update: if request.auth != null && codigo.size() >= 24
        && request.resource.data.keys().hasOnly(['people', 'sobras', 'start', 'slots', 'checked', 'extras']);
    }
  }
}
